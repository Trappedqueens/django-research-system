create table auth_group
(
    id   int auto_increment
        primary key,
    name varchar(150) not null,
    constraint name
        unique (name)
);

create table auth_user
(
    id           int auto_increment
        primary key,
    username     varchar(150)                              not null,
    password     varchar(128)                              not null,
    role         varchar(20)                               not null comment 'admin/teacher/expert',
    college      varchar(100)                              null,
    last_login   datetime(6)                               null,
    is_superuser tinyint(1)   default 0                    not null,
    first_name   varchar(150) default ''                   not null,
    last_name    varchar(150) default ''                   not null,
    email        varchar(254) default ''                   not null,
    is_staff     tinyint(1)   default 0                    not null,
    is_active    tinyint(1)   default 1                    not null,
    date_joined  datetime(6)  default CURRENT_TIMESTAMP(6) not null,
    constraint username
        unique (username)
);

create table django_content_type
(
    id        int auto_increment
        primary key,
    app_label varchar(100) not null,
    model     varchar(100) not null,
    constraint django_content_type_app_label_model_76bd3d3b_uniq
        unique (app_label, model)
);

create table auth_permission
(
    id              int auto_increment
        primary key,
    name            varchar(255) not null,
    content_type_id int          not null,
    codename        varchar(100) not null,
    constraint auth_permission_content_type_id_codename_01ab375a_uniq
        unique (content_type_id, codename),
    constraint auth_permission_content_type_id_2f476e4b_fk_django_co
        foreign key (content_type_id) references django_content_type (id)
);

create table auth_group_permissions
(
    id            bigint auto_increment
        primary key,
    group_id      int not null,
    permission_id int not null,
    constraint auth_group_permissions_group_id_permission_id_0cd325b0_uniq
        unique (group_id, permission_id),
    constraint auth_group_permissio_permission_id_84c5c92e_fk_auth_perm
        foreign key (permission_id) references auth_permission (id),
    constraint auth_group_permissions_group_id_b120cbf9_fk_auth_group_id
        foreign key (group_id) references auth_group (id)
);

create table django_admin_log
(
    id              int auto_increment
        primary key,
    action_time     datetime(6)       not null,
    object_id       longtext          null,
    object_repr     varchar(200)      not null,
    action_flag     smallint unsigned not null,
    change_message  longtext          not null,
    content_type_id int               null,
    user_id         bigint            not null,
    constraint django_admin_log_content_type_id_c4bce8eb_fk_django_co
        foreign key (content_type_id) references django_content_type (id),
    check (`action_flag` >= 0)
);

create table django_migrations
(
    id      bigint auto_increment
        primary key,
    app     varchar(255) not null,
    name    varchar(255) not null,
    applied datetime(6)  not null
);

create table django_session
(
    session_key  varchar(40) not null
        primary key,
    session_data longtext    not null,
    expire_date  datetime(6) not null
);

create index django_session_expire_date_a5c62663
    on django_session (expire_date);

create table research_projects
(
    id           int auto_increment
        primary key,
    title        varchar(255)                  not null comment '项目名称',
    category     varchar(100)                  not null comment '项目类型',
    applicant_id int                           not null comment '申报人ID',
    college      varchar(100)                  not null comment '所属学院',
    budget       decimal(12, 2)                not null comment '经费',
    description  text                          not null comment '项目描述',
    status       varchar(20) default 'pending' null comment 'pending/approved/rejected',
    created_at   datetime                      not null,
    constraint research_projects_ibfk_1
        foreign key (applicant_id) references auth_user (id)
);

create table project_reviews
(
    id          int auto_increment
        primary key,
    project_id  int         not null,
    reviewer_id int         not null comment '专家ID',
    comment     text        not null comment '审核意见',
    result      varchar(20) not null comment 'approved/rejected',
    reviewed_at datetime    not null,
    constraint project_reviews_ibfk_1
        foreign key (project_id) references research_projects (id),
    constraint project_reviews_ibfk_2
        foreign key (reviewer_id) references auth_user (id)
);

create index project_id
    on project_reviews (project_id);

create index reviewer_id
    on project_reviews (reviewer_id);

create index applicant_id
    on research_projects (applicant_id);
